set(WR_SOURCES)
file(GLOB WR_SRCS "webrender/src/*.cpp" "swgl/*.cpp")
add_library(webrender STATIC ${WR_SRCS})
target_include_directories(webrender PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(webrender PUBLIC cxx_std_20)
target_compile_options(webrender PRIVATE -Wall -Wextra -Wno-unused-parameter)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_options(webrender PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(webrender PRIVATE -fsanitize=address,undefined)
endif()
